"use strict";
const { test } = require("node:test");
const assert = require("node:assert/strict");
const fs = require("node:fs");
const os = require("node:os");
const path = require("node:path");
const { randomUUID } = require("node:crypto");
const { io } = require("socket.io-client");
const { createDemoServer } = require("../server");
const { SenderTransfer } = require("../public/transfer");
function event(socket, name, ms = 1500) {
  return new Promise((resolve, reject) => {
    const timer = setTimeout(() => {
      socket.off(name, handler);
      reject(new Error("Event timed out: " + name));
    }, ms);
    const handler = (data) => {
      clearTimeout(timer);
      resolve(data);
    };
    socket.once(name, handler);
  });
}
function ack(socket, name, data) {
  return new Promise((resolve, reject) =>
    socket
      .timeout(1500)
      .emit(name, data, (err, value) => (err ? reject(err) : resolve(value))),
  );
}
test("real HTTP uploads and Socket.IO handoff lifecycle", async (t) => {
  const dir = fs.mkdtempSync(path.join(os.tmpdir(), "flick-test-"));
  const demo = createDemoServer({
    uploadsDir: dir,
    transferTimeout: 220,
    maxFileSize: 2048,
  });
  const clients = [];
  t.after(async () => {
    clients.forEach((s) => s.disconnect());
    await demo.close();
    fs.rmSync(dir, { recursive: true, force: true });
  });
  await new Promise((resolve) => demo.server.listen(0, "127.0.0.1", resolve));
  const base = "http://127.0.0.1:" + demo.server.address().port;
  const connect = async () => {
    const s = io(base, { forceNew: true, reconnectionDelay: 10 });
    clients.push(s);
    await event(s, "connect");
    return s;
  };
  async function upload(bytes, name, type) {
    const form = new FormData();
    form.append("audio", new Blob([bytes], { type }), name);
    return fetch(base + "/api/upload", { method: "POST", body: form });
  }
  const wav = Buffer.alloc(64);
  wav.write("RIFF");
  wav.writeUInt32LE(56, 4);
  wav.write("WAVEfmt ", 8);
  wav.writeUInt32LE(16, 16);
  wav.writeUInt16LE(1, 20);
  wav.writeUInt16LE(1, 22);
  wav.writeUInt32LE(8000, 24);
  wav.writeUInt32LE(16000, 28);
  wav.writeUInt16LE(2, 32);
  wav.writeUInt16LE(16, 34);
  wav.write("data", 36);
  wav.writeUInt32LE(20, 40);
  let track;
  await t.test(
    "safe upload name, URL, range requests and file validation",
    async () => {
      const r = await upload(wav, "../../original.wav", "audio/wav");
      assert.equal(r.status, 201);
      track = await r.json();
      assert.match(track.audioUrl, /^\/uploads\/[a-f0-9-]{36}\.wav$/);
      assert.equal(track.originalName, "original.wav");
      const range = await fetch(base + track.audioUrl, {
        headers: { Range: "bytes=0-9" },
      });
      assert.equal(range.status, 206);
      assert.equal((await range.arrayBuffer()).byteLength, 10);
      assert.equal(
        (await upload("not audio", "text.txt", "text/plain")).status,
        400,
      );
      assert.equal(
        (await upload("not audio", "fake.mp3", "audio/mpeg")).status,
        415,
      );
      assert.equal(
        (await upload(Buffer.alloc(4096), "large.wav", "audio/wav")).status,
        413,
      );
      assert.equal((await fetch(base + "/uploads/missing.mp3")).status, 404);
      assert.equal(
        (
          await fetch(base + "/api/upload", {
            method: "POST",
            body: new FormData(),
          })
        ).status,
        400,
      );
    },
  );
  const a = await connect(),
    aId = randomUUID();
  assert.equal(
    (
      await ack(a, "join-room", {
        role: "sender",
        deviceId: aId,
        room: "demo-room",
      })
    ).ok,
    true,
  );
  const payload = () => ({
    ...track,
    trackName: track.originalName,
    transferId: randomUUID(),
    currentTime: 0.001,
    isPlaying: true,
    senderId: aId,
  });
  const fakeAudio = {
    src: track.audioUrl,
    paused: false,
    ended: false,
    readyState: 1,
    currentTime: 0.001,
    pauseCount: 0,
    getAttribute() {
      return this.src;
    },
    pause() {
      this.pauseCount++;
      this.paused = true;
    },
  };
  const sender = new SenderTransfer({
    audio: fakeAudio,
    socket: a,
    makeId: randomUUID,
    notify() {},
    timeout: 500,
  });
  a.on("transfer-received", (data) => sender.success(data));
  a.on("transfer-failed", (data) => sender.fail(data.transferId, data.message));
  await t.test("no B: request fails and A continues", async () => {
    assert.match(
      (await ack(a, "transfer-music", payload())).message,
      /No receiving/,
    );
    sender.request(track, aId);
    await new Promise((r) => setTimeout(r, 30));
    assert.equal(fakeAudio.pauseCount, 0);
    assert.equal(sender.pending, null);
  });
  const b = await connect(),
    bId = randomUUID();
  const joinB = () =>
    ack(b, "join-room", {
      role: "receiver",
      deviceId: bId,
      room: "demo-room",
      audioReady: true,
    });
  await ack(b, "join-room", {
    role: "receiver",
    deviceId: bId,
    room: "demo-room",
    audioReady: false,
  });
  await t.test("unprepared B: A keeps playing", async () => {
    assert.match(
      (await ack(a, "transfer-music", payload())).message,
      /not ready/,
    );
    assert.equal(fakeAudio.pauseCount, 0);
  });
  await joinB();
  await t.test(
    "success pauses A only after real receiver acknowledgement",
    async () => {
      const incoming = event(b, "transfer-music");
      sender.request(track, aId);
      const data = await incoming;
      assert.equal(data.audioUrl, track.audioUrl);
      assert.equal(fakeAudio.pauseCount, 0);
      assert.equal(
        (
          await ack(a, "transfer-received", {
            transferId: data.transferId,
            currentTime: 0,
          })
        ).ok,
        false,
      );
      const done = event(a, "transfer-received");
      assert.equal(
        (
          await ack(b, "transfer-received", {
            transferId: data.transferId,
            currentTime: 0,
          })
        ).ok,
        true,
      );
      await done;
      assert.equal(fakeAudio.pauseCount, 1);
      fakeAudio.paused = false;
    },
  );
  await t.test(
    "autoplay rejection, duplicate requests and late acknowledgement",
    async () => {
      const incoming = event(b, "transfer-music"),
        p = payload();
      assert.equal((await ack(a, "transfer-music", p)).ok, true);
      await incoming;
      assert.equal((await ack(a, "transfer-music", p)).ok, false);
      assert.equal((await ack(a, "transfer-music", payload())).ok, false);
      const failed = event(a, "transfer-failed");
      b.emit("transfer-failed", {
        transferId: p.transferId,
        message: "Tap to allow playback",
      });
      assert.match((await failed).message, /Tap to allow/);
      assert.equal(
        (
          await ack(b, "transfer-received", {
            transferId: p.transferId,
            currentTime: 0,
          })
        ).ok,
        false,
      );
      assert.equal(fakeAudio.pauseCount, 1);
    },
  );
  await t.test("timeout cancels B without stopping A", async () => {
    const incoming = event(b, "transfer-music"),
      failed = event(a, "transfer-failed"),
      cancelled = event(b, "transfer-cancelled");
    sender.request(track, aId);
    await incoming;
    await failed;
    await cancelled;
    assert.equal(fakeAudio.pauseCount, 1);
    assert.equal(fakeAudio.paused, false);
  });
  await t.test("reconnect re-joins and can transfer again", async () => {
    const connected = event(b, "connect");
    b.io.engine.close();
    await connected;
    assert.equal((await joinB()).ok, true);
    const incoming = event(b, "transfer-music"),
      p = payload();
    assert.equal((await ack(a, "transfer-music", p)).ok, true);
    await incoming;
    const done = event(a, "transfer-received");
    assert.equal(
      (
        await ack(b, "transfer-received", {
          transferId: p.transferId,
          currentTime: 0,
        })
      ).ok,
      true,
    );
    await done;
  });
  await t.test("receiver disconnect cancels pending request", async () => {
    const incoming = event(b, "transfer-music"),
      failed = event(a, "transfer-failed");
    await ack(a, "transfer-music", payload());
    await incoming;
    b.disconnect();
    assert.match((await failed).message, /disconnected/);
    assert.equal(fakeAudio.pauseCount, 1);
  });
  await t.test(
    "deleted file, invalid payload and malformed identity are rejected",
    async () => {
      assert.equal((await ack(a, "transfer-music", null)).ok, false);
      assert.equal((await ack(a, "join-room", null)).ok, false);
      fs.unlinkSync(path.join(dir, path.basename(track.audioUrl)));
      assert.equal((await ack(a, "transfer-music", payload())).ok, false);
    },
  );
});
